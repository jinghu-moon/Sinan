cmake_minimum_required(VERSION 3.24)
project(Sinan VERSION 0.12.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Sinan requires Windows")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Sinan requires a 64-bit toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(CLI11_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CLI11_INSTALL OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
FetchContent_Declare(cli11 URL https://codeload.github.com/CLIUtils/CLI11/tar.gz/refs/tags/v2.5.0 URL_HASH SHA256=17e02b4cddc2fa348e5dbdbb582c59a3486fa2b2433e70a0c3bacb871334fd55)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0 URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187)
FetchContent_Declare(minhook_src URL https://codeload.github.com/TsudaKageyu/minhook/tar.gz/refs/tags/v1.3.4 URL_HASH SHA256=1aebeae4ca898330c507860acc2fca2eb335fe446a3a2b8444c3bf8b2660a14e SOURCE_SUBDIR sinan-no-cmake)
set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FTXUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ftxui URL https://codeload.github.com/ArthurSonzogni/FTXUI/tar.gz/refs/tags/v7.0.3 URL_HASH SHA256=e7c62ffe19009759821b4f0f8df7f2a6fb83784c3a9f1477d81f56d3ee723c88)
option(SINAN_BUILD_TUI "Build interactive terminal consumer" ON)
FetchContent_MakeAvailable(cli11 json minhook_src)
if(SINAN_BUILD_TUI)
  FetchContent_MakeAvailable(ftxui)
endif()
add_library(sinan_decoder STATIC ${minhook_src_SOURCE_DIR}/src/hde/hde64.c)
target_include_directories(sinan_decoder PUBLIC ${minhook_src_SOURCE_DIR}/src/hde)
add_library(sinan_minhook STATIC ${minhook_src_SOURCE_DIR}/src/buffer.c ${minhook_src_SOURCE_DIR}/src/hook.c ${minhook_src_SOURCE_DIR}/src/trampoline.c)
target_link_libraries(sinan_minhook PUBLIC sinan_decoder)
target_include_directories(sinan_minhook PUBLIC ${minhook_src_SOURCE_DIR}/include)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/tomlplusplus/toml.hpp" TOML_HEADER_HASH)
if(NOT TOML_HEADER_HASH STREQUAL "6b5172ad4dd6519aec67b919181fa7a38a2234131e5b2afa232dfe444819783e")
  message(FATAL_ERROR "Vendored toml++ header checksum mismatch")
endif()
add_library(sinan_toml INTERFACE)
target_include_directories(sinan_toml SYSTEM INTERFACE third_party/tomlplusplus)

add_library(sinan_domain STATIC src/policy/display.cpp src/policy/entries.cpp src/policy/rules.cpp src/policy/tab_registry.cpp src/formatter/formatter.cpp)
target_include_directories(sinan_domain PUBLIC src)
target_link_libraries(sinan_domain PUBLIC user32)
add_library(sinan_windows STATIC
  src/platform/windows.cpp src/platform/user_process.cpp src/config/config.cpp src/config/tab_features.cpp
  src/installer/environment.cpp src/platform/explorer.cpp src/platform/shell_tabs.cpp src/platform/shell_collection.cpp
  src/ipc/pipe.cpp src/hook/native.cpp src/hook/controller.cpp src/hook/layout_controller.cpp
  src/hook/features_controller.cpp src/hook/style_controller.cpp src/platform/close_dialog.cpp
  src/compat/image.cpp src/compat/profile.cpp src/compat/feature_contracts.cpp src/compat/xaml.cpp src/compat/prepare.cpp)
target_include_directories(sinan_windows PUBLIC src)
target_compile_definitions(sinan_windows PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX SINAN_VERSION="${PROJECT_VERSION}")
target_link_libraries(sinan_windows PUBLIC sinan_domain sinan_toml sinan_decoder nlohmann_json::nlohmann_json advapi32 shell32 user32 ole32 oleaut32 uuid version bcrypt userenv gdi32 comctl32)
add_library(sinan_client STATIC src/client/agent.cpp)
target_link_libraries(sinan_client PUBLIC sinan_windows)
add_library(sinan_application STATIC src/app/service.cpp src/app/actions.cpp src/installer/install.cpp src/installer/startup.cpp)
target_link_libraries(sinan_application PUBLIC sinan_client)
add_library(sinan_agent_runtime STATIC src/agent/runtime.cpp src/agent/tab_tracker.cpp src/agent/layout_tracker.cpp src/agent/features.cpp src/agent/interaction.cpp src/agent/workspace.cpp)
target_link_libraries(sinan_agent_runtime PUBLIC sinan_windows)
# UI recorder has no FTXUI dependency and can be tested in headless builds.
add_library(sinan_shortcut_recorder STATIC src/ui/shortcut_recorder.cpp)
target_link_libraries(sinan_shortcut_recorder PUBLIC sinan_windows)
if(SINAN_BUILD_TUI)
  add_library(sinan_tui STATIC src/tui/tui.cpp src/tui/i18n.cpp)
  target_link_libraries(sinan_tui PUBLIC sinan_application sinan_shortcut_recorder ftxui::component ftxui::dom ftxui::screen)
endif()
add_executable(sinan_login WIN32 src/installer/login.cpp)
set_target_properties(sinan_login PROPERTIES OUTPUT_NAME sinan-login)
target_compile_definitions(sinan_login PRIVATE UNICODE _UNICODE)
target_link_libraries(sinan_login PRIVATE shell32)
add_executable(sinan_agent WIN32 src/agent/main.cpp)
set_target_properties(sinan_agent PROPERTIES OUTPUT_NAME sinan-agent)
target_link_libraries(sinan_agent PRIVATE sinan_agent_runtime)
add_executable(sinan src/cli/main.cpp)
target_link_libraries(sinan PRIVATE sinan_application CLI11::CLI11)
if(SINAN_BUILD_TUI)
  target_compile_definitions(sinan PRIVATE SINAN_BUILD_TUI=1)
  target_link_libraries(sinan PRIVATE sinan_tui)
endif()
add_executable(sinan_symbols tools/symbol_worker.cpp src/compat/resolver.cpp)
set_target_properties(sinan_symbols PROPERTIES OUTPUT_NAME sinan-symbols)
target_link_libraries(sinan_symbols PRIVATE sinan_windows dbghelp winhttp)
add_library(sinan_hook SHARED src/hook/dll.cpp)
set_target_properties(sinan_hook PROPERTIES OUTPUT_NAME sinan-hook)
target_link_libraries(sinan_hook PRIVATE sinan_windows sinan_minhook)
add_library(sinan_layout SHARED src/hook/layout_dll.cpp src/hook/layout.def)
set_target_properties(sinan_layout PROPERTIES OUTPUT_NAME sinan-layout)
target_link_libraries(sinan_layout PRIVATE sinan_windows runtimeobject)
add_library(sinan_features SHARED src/hook/features_dll.cpp src/hook/features.rc)
set_target_properties(sinan_features PROPERTIES OUTPUT_NAME sinan-features)
target_link_libraries(sinan_features PRIVATE sinan_windows sinan_minhook)
add_library(sinan_style SHARED src/hook/style_dll.cpp src/hook/style.def)
set_target_properties(sinan_style PROPERTIES OUTPUT_NAME sinan-style)
target_link_libraries(sinan_style PRIVATE sinan_windows runtimeobject)
add_dependencies(sinan sinan_symbols sinan_login sinan_agent sinan_hook sinan_layout sinan_features sinan_style)
if(MSVC)
  target_compile_options(sinan_domain PUBLIC /W4 /utf-8 /permissive- /EHsc)
  target_compile_options(sinan_windows PUBLIC /W4 /utf-8 /permissive- /EHsc)
  target_compile_options(sinan_login PRIVATE /W4 /utf-8 /permissive- /EHsc)
endif()
