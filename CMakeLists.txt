if(NOT WIN32)
  message(FATAL_ERROR "TabTail requires Windows")

  message(FATAL_ERROR "TabTail requires a 64-bit toolchain")

  message(FATAL_ERROR "Vendored toml++ header checksum mismatch")

target_include_directories(tabtail_toml SYSTEM INTERFACE third_party/tomlplusplus)

target_sources(tabtail_core PRIVATE src/compat/image.cpp src/compat/profile.cpp src/compat/feature_contracts.cpp src/compat/xaml.cpp src/compat/prepare.cpp)

target_sources(tabtail_core PRIVATE src/agent/layout_tracker.cpp)
target_sources(tabtail_core PRIVATE src/platform/close_dialog.cpp)

target_link_libraries(tabtail_core PUBLIC tabtail_policy tabtail_toml nlohmann_json::nlohmann_json advapi32 shell32 user32 ole32 oleaut32 uuid version bcrypt userenv gdi32 comctl32)

target_link_libraries(tabtail_login PRIVATE shell32)
add_executable(tabtail src/cli/main.cpp)

set_target_properties(tabtail_symbols PROPERTIES OUTPUT_NAME tabtail-symbols)
target_link_libraries(tabtail_symbols PRIVATE tabtail_core dbghelp winhttp)
add_dependencies(tabtail tabtail_symbols)
add_dependencies(tabtail tabtail_login)
target_link_libraries(tabtail PRIVATE tabtail_core tabtail_tui CLI11::CLI11)
add_library(tabtail_hook SHARED src/hook/dll.cpp)
set_target_properties(tabtail_hook PROPERTIES OUTPUT_NAME tabtail-hook)
target_link_libraries(tabtail_hook PRIVATE tabtail_core tabtail_minhook)
add_dependencies(tabtail tabtail_hook)
add_library(tabtail_layout SHARED src/hook/layout_dll.cpp src/hook/layout.def)
set_target_properties(tabtail_layout PROPERTIES OUTPUT_NAME tabtail-layout)
target_link_libraries(tabtail_layout PRIVATE tabtail_core runtimeobject)
add_dependencies(tabtail tabtail_layout)

set_target_properties(tabtail_features PROPERTIES OUTPUT_NAME tabtail-features)
target_link_libraries(tabtail_features PRIVATE tabtail_core tabtail_minhook)
add_dependencies(tabtail tabtail_features)
